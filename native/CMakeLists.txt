cmake_minimum_required(VERSION 3.20)
project(VoiceLockNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
set(CMAKE_POLICY_DEFAULT_CMP0115 OLD)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory(deps/soxr EXCLUDE_FROM_ALL)
add_executable(VoiceLock.AudioEngine src/main.cpp)
target_include_directories(VoiceLock.AudioEngine PRIVATE deps/ort deps/json deps/soxr/src)
target_compile_definitions(VoiceLock.AudioEngine PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN ORT_API_MANUAL_INIT)
# Ring cursors deliberately occupy separate cache lines (C4324).
target_compile_options(VoiceLock.AudioEngine PRIVATE /W4 /wd4324 /EHsc /utf-8)
target_link_libraries(VoiceLock.AudioEngine PRIVATE soxr ole32 uuid avrt bcrypt psapi advapi32 winhttp)
enable_testing()
add_test(NAME native_core COMMAND VoiceLock.AudioEngine selftest)
